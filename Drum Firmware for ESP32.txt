
#include <Wire.h>
#include <Adafruit_MPR121.h>

Adafruit_MPR121 touch = Adafruit_MPR121();

const int SDA_PIN = 8;
const int SCL_PIN = 9;
const int LED_PIN = 4;

uint16_t previousTouches = 0;
bool recording = false;

void setup() {
  Serial.begin(115200);
  pinMode(LED_PIN, OUTPUT);
  digitalWrite(LED_PIN, LOW);

  Wire.begin(SDA_PIN, SCL_PIN);

  if (!touch.begin(0x5A)) {
    while (true) {
      delay(100);
    }
  }
}

void loop() {
  uint16_t currentTouches = touch.touched();
  uint16_t newTouches = currentTouches & ~previousTouches;

  for (int i = 0; i < 12; i++) {
    if (newTouches & (1 << i)) {
      if (i < 9) {
        Serial.print("PAD");
        Serial.println(i + 1);
      } else if (i == 9) {
        recording = !recording;
        digitalWrite(LED_PIN, recording ? HIGH : LOW);
        Serial.println("RECORD");
      } else if (i == 10) {
        Serial.println("LOOP");
      } else if (i == 11) {
        Serial.println("DISCARD");
      }
    }
  }

  previousTouches = currentTouches;
  delay(10);
}
